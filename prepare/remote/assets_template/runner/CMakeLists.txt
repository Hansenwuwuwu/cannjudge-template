cmake_minimum_required(VERSION 3.16)
project(current_audit_runner LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CANN_ROOT /usr/local/Ascend/cann-9.0.0)
add_executable(current_audit_runner main.cpp)
target_compile_options(current_audit_runner PRIVATE -O2 -Wall -Wextra -Werror)
target_include_directories(current_audit_runner PRIVATE "${CANN_ROOT}/aarch64-linux/include" "${CURRENT_AUDIT_INCLUDE}")
target_link_libraries(current_audit_runner PRIVATE "${CANN_ROOT}/lib64/libnnopbase.so" "${CANN_ROOT}/lib64/libascendcl.so" ${CMAKE_DL_LIBS})
set_target_properties(current_audit_runner PROPERTIES BUILD_RPATH "${CANN_ROOT}/lib64")
